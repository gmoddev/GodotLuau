cmake_minimum_required(VERSION 3.20)
project(NativeSanitizerControls LANGUAGES CXX)
option(GL_SANITIZE_HOST "Instrument the standalone library-loader control" ON)
option(GL_SANITIZE_LIBRARY "Instrument the minimal GDExtension control" ON)
set(GL_GDEXTENSION_HEADER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../godot-cpp/gdextension" CACHE PATH "Pinned GDExtension C interface headers")
if(NOT EXISTS "${GL_GDEXTENSION_HEADER_ROOT}/gdextension_interface.h")
    message(FATAL_ERROR "Pinned GDExtension interface header is required")
endif()
if(MSVC)
    # MSVC ASan cannot be combined with CMake's default /RTC debug checks.
    string(REGEX REPLACE "/RTC[^ ]*" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDebugDLL")
endif()
add_library(NativeSanitizerProbe SHARED NativeSanitizerProbe.cpp)
target_include_directories(NativeSanitizerProbe PRIVATE "${GL_GDEXTENSION_HEADER_ROOT}")
set_target_properties(NativeSanitizerProbe PROPERTIES PREFIX "")
add_executable(NativeSanitizerLoader NativeSanitizerLoader.cpp)
target_link_libraries(NativeSanitizerLoader PRIVATE ${CMAKE_DL_LIBS})
foreach(Target NativeSanitizerProbe NativeSanitizerLoader)
    target_compile_features(${Target} PRIVATE cxx_std_17)
    if((Target STREQUAL "NativeSanitizerProbe" AND GL_SANITIZE_LIBRARY) OR
       (Target STREQUAL "NativeSanitizerLoader" AND GL_SANITIZE_HOST))
        if(MSVC)
            target_compile_options(${Target} PRIVATE /fsanitize=address /Zi /Od)
            target_link_options(${Target} PRIVATE /INCREMENTAL:NO)
        else()
            target_compile_options(${Target} PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
            target_link_options(${Target} PRIVATE -fsanitize=address)
        endif()
    endif()
endforeach()
