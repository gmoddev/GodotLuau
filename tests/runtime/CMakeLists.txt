cmake_minimum_required(VERSION 3.20)
project(GodotLuauRuntimeTests LANGUAGES CXX)
enable_testing()
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
add_subdirectory(../../luau ${CMAKE_CURRENT_BINARY_DIR}/luau EXCLUDE_FROM_ALL)
add_executable(ExecutionDomainTests ExecutionDomainTests.cpp ../../src/core/gl_execution_domain.cpp)
target_compile_features(ExecutionDomainTests PRIVATE cxx_std_17)
target_include_directories(ExecutionDomainTests PRIVATE ../../src/core)
target_link_libraries(ExecutionDomainTests PRIVATE Luau.Compiler Luau.VM)
if(MSVC)
    target_compile_options(ExecutionDomainTests PRIVATE /W4 /EHsc)
endif()
add_test(NAME ExecutionDomains COMMAND ExecutionDomainTests)
set_tests_properties(ExecutionDomains PROPERTIES TIMEOUT 30)

# Optional build of the actual Godot adapter against the same pinned sources.
# SConstruct remains the existing distribution build entry point.
option(GL_BUILD_GODOT_ADAPTER "Compile the full extension for integration fixtures" OFF)
if(GL_BUILD_GODOT_ADAPTER)
    add_subdirectory(../../godot-cpp ${CMAKE_CURRENT_BINARY_DIR}/godot-cpp)
    file(GLOB CoreSources CONFIGURE_DEPENDS ../../src/core/*.cpp)
    add_library(GodotLuauAdapter SHARED ${CoreSources})
    target_compile_features(GodotLuauAdapter PRIVATE cxx_std_17)
    target_link_libraries(GodotLuauAdapter PRIVATE godot::cpp Luau.Compiler Luau.VM)
    target_include_directories(GodotLuauAdapter PRIVATE
        ../../src/core ../../src/services ../../src/characters ../../src/gameplay
        ../../src/ui ../../src/editor ../../src/importer)
    if(MSVC)
        target_compile_options(GodotLuauAdapter PRIVATE /EHsc /bigobj)
    endif()
    set_target_properties(GodotLuauAdapter PROPERTIES PREFIX "" OUTPUT_NAME godot_luau)
endif()
