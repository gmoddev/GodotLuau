# Optional build of the actual Godot adapter against the same pinned sources.
# SConstruct remains the existing distribution build entry point.

set(GL_BASELINE_SOURCE_ROOT "" CACHE PATH "Optional inherited source tree for controlled benchmarks")
option(GL_BUILD_DEBUGGER_BENCHMARKS "Measure production-core debugger modes against the existing baseline" OFF)
if(GL_BUILD_DEBUGGER_BENCHMARKS)
    foreach(Variant Candidate Baseline)
        if(Variant STREQUAL "Candidate")
            set(Root ${GL_SOURCE_ROOT})
        else()
            set(Root ${GL_BASELINE_SOURCE_ROOT})
        endif()
        if(NOT Root)
            message(FATAL_ERROR "Debugger benchmark baseline is required")
        endif()
        set(Sources DebuggerBenchmarks.cpp ${Root}/src/core/gl_execution_domain.cpp)
        if(Variant STREQUAL "Candidate")
            list(APPEND Sources ${Root}/src/core/gl_debugger.cpp)
        endif()
        add_executable(DebuggerBenchmark${Variant} ${Sources})
        target_compile_features(DebuggerBenchmark${Variant} PRIVATE cxx_std_17)
        target_include_directories(DebuggerBenchmark${Variant} PRIVATE ${Root}/src/core)
        target_link_libraries(DebuggerBenchmark${Variant} PRIVATE Luau.Compiler Luau.VM)
        if(Variant STREQUAL "Candidate")
            target_compile_definitions(DebuggerBenchmark${Variant} PRIVATE GL_DEBUGGER_BENCHMARK)
        endif()
        if(MSVC)
            target_compile_options(DebuggerBenchmark${Variant} PRIVATE /EHsc)
        endif()
    endforeach()
endif()
if(GL_BUILD_GODOT_ADAPTER)
    add_subdirectory(../../godot-cpp ${CMAKE_CURRENT_BINARY_DIR}/godot-cpp)
    function(GLAdapter Target Root Output)
        file(GLOB CoreSources CONFIGURE_DEPENDS ${Root}/src/core/*.cpp)
        add_library(${Target} SHARED ${CoreSources})
        target_compile_features(${Target} PRIVATE cxx_std_17)
        target_link_libraries(${Target} PRIVATE godot::cpp Luau.Compiler Luau.VM)
        target_include_directories(${Target} PRIVATE
            ${Root}/src/core ${Root}/src/services ${Root}/src/characters ${Root}/src/gameplay
            ${Root}/src/ui ${Root}/src/editor ${Root}/src/importer)
        if(MSVC)
            target_compile_options(${Target} PRIVATE /EHsc /bigobj)
        endif()
        set_target_properties(${Target} PROPERTIES PREFIX "" OUTPUT_NAME ${Output})
    endfunction()
    GLAdapter(GodotLuauAdapter ${GL_SOURCE_ROOT} godot_luau)
    add_dependencies(GodotLuauAdapter DataStoreWorker)
    target_link_libraries(GodotLuauAdapter PRIVATE ${CMAKE_DL_LIBS})
    # Teardown access for the native fixture; never part of the SCons product.
    target_compile_definitions(GodotLuauAdapter PRIVATE GL_RUNTIME_TESTS)
    if(GL_BASELINE_SOURCE_ROOT)
        GLAdapter(GodotLuauBaselineAdapter ${GL_BASELINE_SOURCE_ROOT} godot_luau_baseline)
    endif()
endif()
