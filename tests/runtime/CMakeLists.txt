cmake_minimum_required(VERSION 3.20)
project(GodotLuauRuntimeTests LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
enable_testing()
set(GL_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Production source tree; baseline builds use the same pinned dependencies")
option(GL_BUILD_CORE_TESTS "Build shared-domain core regressions" ON)
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
add_subdirectory(../../luau ${CMAKE_CURRENT_BINARY_DIR}/luau EXCLUDE_FROM_ALL)
if(GL_BUILD_CORE_TESTS)
add_executable(ExecutionDomainTests ExecutionDomainTests.cpp ${GL_SOURCE_ROOT}/src/core/gl_execution_domain.cpp)
target_compile_features(ExecutionDomainTests PRIVATE cxx_std_17)
target_include_directories(ExecutionDomainTests PRIVATE ${GL_SOURCE_ROOT}/src/core)
target_link_libraries(ExecutionDomainTests PRIVATE Luau.Compiler Luau.VM)
if(MSVC)
    target_compile_options(ExecutionDomainTests PRIVATE /W4 /EHsc)
endif()
add_test(NAME ExecutionDomains COMMAND ExecutionDomainTests)
set_tests_properties(ExecutionDomains PROPERTIES TIMEOUT 30)
option(GL_BUILD_HOT_PATH_BENCHMARKS "Build production-core allocation/CPU microbenchmarks" OFF)
if(GL_BUILD_HOT_PATH_BENCHMARKS)
    add_executable(HotPathBenchmarks HotPathBenchmarks.cpp ${GL_SOURCE_ROOT}/src/core/gl_execution_domain.cpp)
    target_compile_features(HotPathBenchmarks PRIVATE cxx_std_17)
    target_include_directories(HotPathBenchmarks PRIVATE ${GL_SOURCE_ROOT}/src/core)
    target_link_libraries(HotPathBenchmarks PRIVATE Luau.Compiler Luau.VM)
    if(MSVC)
        target_compile_options(HotPathBenchmarks PRIVATE /EHsc)
    endif()
endif()
endif()

# Optional build of the actual Godot adapter against the same pinned sources.
# SConstruct remains the existing distribution build entry point.
option(GL_BUILD_GODOT_ADAPTER "Compile the full extension for integration fixtures" OFF)
set(GL_BASELINE_SOURCE_ROOT "" CACHE PATH "Optional inherited source tree for controlled benchmarks")
if(GL_BUILD_GODOT_ADAPTER)
    add_subdirectory(../../godot-cpp ${CMAKE_CURRENT_BINARY_DIR}/godot-cpp)
    function(GLAdapter Target Root Output)
        file(GLOB CoreSources CONFIGURE_DEPENDS ${Root}/src/core/*.cpp)
        add_library(${Target} SHARED ${CoreSources})
        target_compile_features(${Target} PRIVATE cxx_std_17)
        target_link_libraries(${Target} PRIVATE godot::cpp Luau.Compiler Luau.VM)
        target_include_directories(${Target} PRIVATE
            ${Root}/src/core ${Root}/src/services ${Root}/src/characters ${Root}/src/gameplay
            ${Root}/src/ui ${Root}/src/editor ${Root}/src/importer)
        if(MSVC)
            target_compile_options(${Target} PRIVATE /EHsc /bigobj)
        endif()
        set_target_properties(${Target} PROPERTIES PREFIX "" OUTPUT_NAME ${Output})
    endfunction()
    GLAdapter(GodotLuauAdapter ${GL_SOURCE_ROOT} godot_luau)
    # Teardown access for the native fixture; never part of the SCons product.
    target_compile_definitions(GodotLuauAdapter PRIVATE GL_RUNTIME_TESTS)
    if(GL_BASELINE_SOURCE_ROOT)
        GLAdapter(GodotLuauBaselineAdapter ${GL_BASELINE_SOURCE_ROOT} godot_luau_baseline)
    endif()
endif()
